const DEFAULT_STATE = {
    sites: [],
    currentIndex: 0,
    status: 'idle',
    log: [],
    userCredentials: {
        email: '',
        firstName: '',
        lastName: '',
        phone: '',
        zipCode: '',
        customMessage: '',
        jobTitle: '',
        companyName: '',
        websiteUrl: '',
    },
    processingTabId: null,
    siteTimeout: 15000,
    enableScrolling: true,
    processingUrl: null,
    isPaused: false,
};

let state = { ...DEFAULT_STATE };
let processingTimeout = null;
let tabUpdatedListener = null;

async function loadStateFromStorage() {
    const storedState = await chrome.storage.local.get(['extensionState']);
    if (storedState.extensionState) {
        state = { ...state, ...storedState.extensionState };
        if (!state.userCredentials) state.userCredentials = { ...DEFAULT_STATE.userCredentials };
        if (state.userCredentials.jobTitle === undefined) state.userCredentials.jobTitle = '';
        if (state.userCredentials.companyName === undefined) state.userCredentials.companyName = '';
        if (state.userCredentials.websiteUrl === undefined) state.userCredentials.websiteUrl = '';
        console.log('State loaded from storage.');
    }
}

function saveStateToStorage() {
    chrome.storage.local.set({ extensionState: state }).catch(err => console.error("Error saving state:", err));
}

function updateStateAndNotify(changes) {
    Object.assign(state, changes);
    saveStateToStorage();
    chrome.runtime.sendMessage({ action: "UPDATE_UI", payload: state }).catch(() => {});
}

function addLog(message, type = 'info') {
    state.log.unshift({ message, type, timestamp: new Date().toLocaleTimeString() });
    if (state.log.length > 100) state.log.pop();
    updateStateAndNotify({});
}

function closeProcessingTab() {
    if (tabUpdatedListener) {
        chrome.tabs.onUpdated.removeListener(tabUpdatedListener);
        tabUpdatedListener = null;
    }
    if (state.processingTabId !== null) {
        chrome.tabs.remove(state.processingTabId).catch(() => {});
        state.processingTabId = null;
    }
    if (processingTimeout) {
        clearTimeout(processingTimeout);
        processingTimeout = null;
    }
    state.processingUrl = null;
}

async function processNextSite() {
    if (state.status === 'paused') return;
    if (state.status !== 'running') return;

    if (state.currentIndex >= state.sites.length) {
        addLog('All sites have been processed.', 'success');
        updateStateAndNotify({ status: 'completed' });
        closeProcessingTab();
        return;
    }

    const url = state.sites[state.currentIndex];

    if (state.processingUrl === url) {
        console.log('URL already being processed, skipping:', url);
        moveToNextSite();
        return;
    }

    state.processingUrl = url;
    addLog(`(${state.currentIndex + 1}/${state.sites.length}) Processing: ${url}`, 'info');
    updateStateAndNotify({});

    chrome.tabs.create({ url: url, active: false }, (tab) => {
        if (!tab || !tab.id) {
            addLog(`Error creating tab for ${url}`, 'error');
            moveToNextSite();
            return;
        }

        state.processingTabId = tab.id;

        processingTimeout = setTimeout(() => {
            addLog(`Timeout for ${url} after ${state.siteTimeout/1000}s, moving to next site.`, 'warning');
            moveToNextSite();
        }, state.siteTimeout);

        tabUpdatedListener = (tabId, changeInfo, tab) => {
            if (tabId === state.processingTabId && changeInfo.status === 'complete') {
                chrome.tabs.onUpdated.removeListener(tabUpdatedListener);
                tabUpdatedListener = null;

                if (tab && tab.url && (tab.url.startsWith('chrome://') || tab.url.startsWith('about:'))) {
                    addLog(`Tab showing error page: ${tab.url}`, 'error');
                    moveToNextSite();
                    return;
                }

                setTimeout(() => {
                    if (state.status !== 'running') return;

                    chrome.scripting.executeScript({ target: { tabId: tabId }, files: ['content.js'] }, () => {
                        if (chrome.runtime.lastError) {
                            addLog(`Error injecting script: ${chrome.runtime.lastError.message}`, 'error');
                            moveToNextSite();
                        } else {
                            if (processingTimeout) {
                                clearTimeout(processingTimeout);
                                processingTimeout = setTimeout(() => {
                                    addLog(`Form processing timeout for ${url}, moving to next site.`, 'warning');
                                    moveToNextSite();
                                }, 12000);
                            }

                            chrome.tabs.sendMessage(tabId, {
                                action: "AUTOFILL_FORM",
                                data: state.userCredentials,
                                enableScrolling: state.enableScrolling
                            });
                        }
                    });
                }, 5000);
            }
        };

        chrome.tabs.onUpdated.addListener(tabUpdatedListener);
    });
}

function moveToNextSite() {
    closeProcessingTab();
    state.currentIndex++;
    updateStateAndNotify({});
    setTimeout(() => {
        processNextSite();
    }, 3000);
}

chrome.runtime.onMessage.addListener((request, sender, sendResponse) => {
    try {
        switch (request.action) {
            case "GET_STATE":
                sendResponse(state);
                break;
            case "SAVE_CREDENTIALS":
                state.userCredentials = { ...DEFAULT_STATE.userCredentials, ...request.data };
                saveStateToStorage();
                sendResponse({ success: true });
                break;
            case "CLEAR_STATE":
                state = { ...DEFAULT_STATE };
                saveStateToStorage();
                closeProcessingTab();
                sendResponse({ success: true });
                break;
            case "UPDATE_TIMEOUT":
                if (request.timeout && request.timeout >= 5000) {
                    state.siteTimeout = request.timeout;
                    saveStateToStorage();
                    sendResponse({ success: true });
                } else {
                    sendResponse({ success: false, error: 'Timeout must be at least 5000ms' });
                }
                break;
            case "UPDATE_SCROLL_SETTING":
                state.enableScrolling = request.enableScrolling;
                saveStateToStorage();
                sendResponse({ success: true });
                break;
            case "START_PROCESSING":
                if (state.status === 'running' || state.status === 'paused') {
                    sendResponse({ error: "Already processing or paused." });
                    return;
                }
                state.sites = request.sites.map(url => url.startsWith('http') ? url : 'https://' + url);
                state.userCredentials = { ...DEFAULT_STATE.userCredentials, ...request.data };
                state.enableScrolling = request.enableScrolling !== undefined ? request.enableScrolling : state.enableScrolling;
                state.currentIndex = 0;
                state.processingUrl = null;
                state.isPaused = false;
                state.log = [];
                addLog('Process started.', 'success');
                updateStateAndNotify({ status: 'running' });
                processNextSite();
                sendResponse({ success: true });
                break;
            case "PAUSE_PROCESSING":
                if (state.status === 'running') {
                    state.isPaused = true;
                    addLog('Process paused.', 'warning');
                    updateStateAndNotify({ status: 'paused' });
                    sendResponse({ success: true });
                } else {
                    sendResponse({ success: false, error: "Not currently running" });
                }
                break;
            case "STOP_PROCESSING":
                if (state.status !== 'idle' && state.status !== 'stopped') {
                    addLog('Process stopped by user.', 'error');
                    updateStateAndNotify({ status: 'stopped', currentIndex: 0, sites: [], processingUrl: null, isPaused: false });
                    closeProcessingTab();
                    sendResponse({ success: true });
                } else {
                    sendResponse({ success: false, error: "Not currently running" });
                }
                break;
            case "RESUME_PROCESSING":
                if (state.status === 'paused') {
                    state.isPaused = false;
                    addLog('Process resumed.', 'success');
                    updateStateAndNotify({ status: 'running' });
                    processNextSite();
                    sendResponse({ success: true });
                } else {
                    sendResponse({ success: false, error: "Not currently paused" });
                }
                break;
        }
    } catch (e) {
        console.error('Error handling message:', e);
        sendResponse({ success: false, error: e.message });
    }
    return true;
});

            websiteUrl: websiteUrlInput.value.trim(),
        };
        chrome.runtime.sendMessage({ action: "SAVE_CREDENTIALS", data }, (response) => {
            if (response && response.success) {
                saveInfoBtn.textContent = 'Saved!';
                saveInfoBtn.style.background = 'linear-gradient(135deg, #28a745, #20c997)';
                setTimeout(() => {
                    saveInfoBtn.textContent = 'Save Info';
                    saveInfoBtn.style.background = '';
                }, 1500);
            }
        });
    });

    startBtn.addEventListener('click', () => {
        const email = emailInput.value.trim();
        if (!email) {
            emailInput.style.borderColor = '#dc3545';
            emailInput.focus();
            setTimeout(() => { emailInput.style.borderColor = ''; }, 2000);
            return;
        }
        const sites = siteList.value.split('\n').map(url => url.trim()).filter(url => url.length > 0);
        if (sites.length === 0) {
            siteList.style.borderColor = '#dc3545';
            siteList.focus();
            setTimeout(() => { siteList.style.borderColor = ''; }, 2000);
            return;
        }
        const data = {
            email: email,
            firstName: firstNameInput.value.trim(),
            lastName: lastNameInput.value.trim(),
            phone: phoneInput.value.trim(),
            zipCode: zipCodeInput.value.trim(),
            customMessage: customMessageInput.value.trim(),
            jobTitle: jobTitleInput.value.trim(),
            companyName: companyNameInput.value.trim(),
            websiteUrl: websiteUrlInput.value.trim(),
        };
        chrome.runtime.sendMessage({
            action: "START_PROCESSING",
            sites,
            data,
            enableScrolling: scrollCheckbox.checked
        }, (response) => {
            if (response && response.success) updateButtonStates('running');
            else if (response && response.error) alert(response.error);
        });
    });

    pauseBtn.addEventListener('click', () => {
        chrome.runtime.sendMessage({ action: "PAUSE_PROCESSING" }, (response) => {
            if (response && response.success) updateButtonStates('paused');
        });
    });

    resumeBtn.addEventListener('click', () => {
        chrome.runtime.sendMessage({ action: "RESUME_PROCESSING" }, (response) => {
            if (response && response.success) updateButtonStates('running');
        });
    });

    stopBtn.addEventListener('click', () => {
        if (confirm('Are you sure you want to stop processing?')) {
            chrome.runtime.sendMessage({ action: "STOP_PROCESSING" }, (response) => {
                if (response && response.success) updateButtonStates('stopped');
            });
        }
    });

    clearBtn.addEventListener('click', () => {
        if (confirm('Clear all data and reset?')) {
            chrome.runtime.sendMessage({ action: "CLEAR_STATE" }, (response) => {
                if (response && response.success) {
                    emailInput.value = '';
                    firstNameInput.value = '';
                    lastNameInput.value = '';
                    phoneInput.value = '';
                    zipCodeInput.value = '';
                    customMessageInput.value = '';
                    jobTitleInput.value = '';
                    companyNameInput.value = '';
                    websiteUrlInput.value = '';
                    siteList.value = '';
                    logContainer.innerHTML = '';
                    progressBar.value = 0;
                    progressBar.max = 0;
                    progressText.textContent = '0/0';
                    updateButtonStates('idle');
                }
            });
        }
    });

    timeoutSlider.addEventListener('input', () => {
        const value = parseInt(timeoutSlider.value);
        timeoutValue.textContent = value;
        chrome.runtime.sendMessage({ action: "UPDATE_TIMEOUT", timeout: value * 1000 });
    });

    scrollCheckbox.addEventListener('change', () => {
        chrome.runtime.sendMessage({ action: "UPDATE_SCROLL_SETTING", enableScrolling: scrollCheckbox.checked });
    });

    function updateButtonStates(status) {
        startBtn.disabled = status === 'running' || status === 'paused';
        pauseBtn.disabled = status !== 'running';
        resumeBtn.disabled = status !== 'paused';
        stopBtn.disabled = status === 'idle' || status === 'stopped';
        statusIndicator.className = 'status-indicator';
        if (status === 'running') statusIndicator.classList.add('running');
        else if (status === 'paused') statusIndicator.classList.add('paused');
        else if (status === 'stopped') statusIndicator.classList.add('stopped');
        else if (status === 'completed') statusIndicator.classList.add('completed');
    }

    function updateUI(state) {
        if (state.sites && state.sites.length > 0) {
            progressBar.max = state.sites.length;
            progressBar.value = state.currentIndex || 0;
            progressText.textContent = `${state.currentIndex || 0}/${state.sites.length}`;
        } else {
            progressBar.max = 0;
            progressBar.value = 0;
            progressText.textContent = '0/0';
        }
        updateButtonStates(state.status);
        renderLog(state.log);
        if (state.userCredentials) {
            if (state.userCredentials.email && !emailInput.value) emailInput.value = state.userCredentials.email;
            if (state.userCredentials.firstName && !firstNameInput.value) firstNameInput.value = state.userCredentials.firstName;
            if (state.userCredentials.lastName && !lastNameInput.value) lastNameInput.value = state.userCredentials.lastName;
            if (state.userCredentials.phone && !phoneInput.value) phoneInput.value = state.userCredentials.phone;
            if (state.userCredentials.zipCode && !zipCodeInput.value) zipCodeInput.value = state.userCredentials.zipCode;
            if (state.userCredentials.customMessage && !customMessageInput.value) customMessageInput.value = state.userCredentials.customMessage;
            if (state.userCredentials.jobTitle && !jobTitleInput.value) jobTitleInput.value = state.userCredentials.jobTitle;
            if (state.userCredentials.companyName && !companyNameInput.value) companyNameInput.value = state.userCredentials.companyName;
            if (state.userCredentials.websiteUrl && !websiteUrlInput.value) websiteUrlInput.value = state.userCredentials.websiteUrl;
        }
        if (state.siteTimeout) {
            const seconds = Math.round(state.siteTimeout / 1000);
            timeoutSlider.value = seconds;
            timeoutValue.textContent = seconds;
        }
        if (state.enableScrolling !== undefined) scrollCheckbox.checked = state.enableScrolling;
    }

    function renderLog(log) {
        if (!log || log.length === 0) {
            logContainer.innerHTML = '<div style="color:#999;text-align:center;padding:10px;">No activity yet</div>';
            return;
        }
        logContainer.innerHTML = log.map(entry => `
            <div class="log-entry ${entry.type || 'info'}">
                <span class="timestamp">${entry.timestamp || ''}</span>
                <span class="message">${escapeHtml(entry.message || '')}</span>
            </div>
        `).join('');
        logContainer.scrollTop = 0;
    }

    function escapeHtml(text) {
        const div = document.createElement('div');
        div.textContent = text;
        return div.innerHTML;
    }

    chrome.runtime.onMessage.addListener((request, sender, sendResponse) => {
        if (request.action === "UPDATE_UI") updateUI(request.payload);
    });

    chrome.runtime.sendMessage({ action: "GET_STATE" }, (state) => {
        if (state) updateUI(state);
    });
});
